lemma encoding_prefix_write:
    parameter(prefix, words, type(list(Z)))
    parameter(x, type(_ (inferred)))
    statement:
        arrow:
            in_scope nat(less_than(length(prefix), length(words)))
            apply:
                notation "<[ _ := _ ]>":
                    length(prefix)
                    x
                append(prefix, drop(length(prefix), words))
            =append(append(prefix, list(x)), drop(S(length(prefix)), words))

Depends on

No resolved project dependencies.

Original declaration (proof-free)
Lemma encoding_prefix_write (prefix words : list Z) x :
 (length prefix < length words)%nat ->
 <[length prefix := x]> (prefix ++ drop (length prefix) words) =
 (prefix ++ [x]) ++ drop (S (length prefix)) words.
