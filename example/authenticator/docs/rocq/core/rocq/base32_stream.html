Lemma read_symbols_snoc chars ch ds pad :
 read_symbols chars = Some (ds,pad) ->
 read_symbols (chars ++ [ch]) =
 if ch =? 61 then Some (ds,S pad) else
 if (pad =? 0)%nat then
